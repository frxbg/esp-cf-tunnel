cmake_minimum_required(VERSION 3.20)
project(idf_tls_regression C)
enable_testing()
set(CMAKE_C_STANDARD 11)
foreach(variant baseline patched)
    add_executable(tls_${variant} test_connect.c)
    target_include_directories(tls_${variant} PRIVATE "${SDK_EXTRACT_DIR}")
    target_compile_definitions(tls_${variant} PRIVATE SDK_EXTRACT="${variant}.inc")
    if(variant STREQUAL "patched")
        target_compile_definitions(tls_${variant} PRIVATE EXPECT_PATCHED=1)
    endif()
    if(NOT MSVC)
        target_compile_options(tls_${variant} PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(tls_${variant} PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME sdk_${variant} COMMAND tls_${variant})
endforeach()
