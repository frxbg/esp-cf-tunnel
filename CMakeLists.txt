cmake_minimum_required(VERSION 3.20)
project(esp_cf_tunnel_host VERSION 0.3.2 LANGUAGES C)
include(CTest)
option(CF_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer (Clang/GCC)" OFF)
option(CF_FUZZING "Build libFuzzer parser harness (Clang)" OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CF_COMPONENT ${CMAKE_CURRENT_SOURCE_DIR}/components/esp_cf_tunnel)
include(${CF_COMPONENT}/core_sources.cmake)
list(TRANSFORM CF_CORE_SOURCES PREPEND "${CF_COMPONENT}/")
add_library(cf_json_vendor STATIC ${CF_COMPONENT}/third_party/cjson/cf_cJSON.c)
target_include_directories(cf_json_vendor PUBLIC ${CF_COMPONENT}/third_party/cjson)
target_compile_definitions(cf_json_vendor PRIVATE CJSON_NESTING_LIMIT=16)
add_library(cf_core STATIC ${CF_CORE_SOURCES})
include(${CF_COMPONENT}/third_party/nghttp2/library.cmake)
add_library(cf_nghttp2 STATIC ${CF_NGHTTP2_SOURCES})
cf_configure_nghttp2(cf_nghttp2)
add_library(cf_h2 STATIC ${CF_COMPONENT}/src/transport/cf_h2.c)
target_include_directories(cf_h2 PUBLIC ${CF_COMPONENT}/include)
target_link_libraries(cf_h2 PUBLIC cf_core PRIVATE cf_nghttp2)
target_include_directories(cf_core PUBLIC ${CF_COMPONENT}/include PRIVATE ${CF_COMPONENT}/src/protocol)
target_link_libraries(cf_core PRIVATE cf_json_vendor)
if(MSVC)
    target_compile_options(cf_core PRIVATE /W4 /WX)
    target_compile_options(cf_h2 PRIVATE /W4 /WX)
else()
    target_compile_options(cf_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(cf_h2 PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(cf_json_vendor PUBLIC m)
endif()
if(CF_SANITIZERS OR CF_FUZZING)
    if(MSVC)
        message(FATAL_ERROR "Use Linux/WSL GCC or Clang for the ASan+UBSan profile")
    endif()
    foreach(target cf_core cf_json_vendor cf_h2 cf_nghttp2)
        target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endforeach()
endif()
if(BUILD_TESTING)
    add_executable(cf_tls_diag_tests tests/host/test_tls_diag.c ${CF_COMPONENT}/src/transport/cf_tls_diag.c)
    target_include_directories(cf_tls_diag_tests PRIVATE tests/tls_stubs ${CF_COMPONENT}/include ${CF_COMPONENT}/src/transport)
    target_link_libraries(cf_tls_diag_tests PRIVATE cf_core)
    add_test(NAME tls_diagnostics COMMAND cf_tls_diag_tests)
    add_executable(cf_core_tests tests/host/test_core.c)
    target_link_libraries(cf_core_tests PRIVATE cf_core)
    target_include_directories(cf_core_tests PRIVATE tests/fixtures)
    foreach(suite base64 headers credentials framing config lifecycle dns)
        add_test(NAME ${suite} COMMAND cf_core_tests ${suite})
    endforeach()
    add_executable(cf_rpc_tests tests/host/test_rpc.c)
    target_link_libraries(cf_rpc_tests PRIVATE cf_core)
    target_include_directories(cf_rpc_tests PRIVATE tests/fixtures/rpc)
    add_test(NAME rpc COMMAND cf_rpc_tests)
    add_executable(cf_h2_tests tests/host/test_h2.c)
    target_link_libraries(cf_h2_tests PRIVATE cf_h2 cf_nghttp2)
    add_test(NAME http2 COMMAND cf_h2_tests)
    add_executable(monitor_remote_tests tests/host/test_monitor_remote.c examples/system_monitor/main/monitor_remote.c)
    target_include_directories(monitor_remote_tests PRIVATE examples/system_monitor/main)
    target_link_libraries(monitor_remote_tests PRIVATE cf_h2)
    add_test(NAME monitor_remote COMMAND monitor_remote_tests)
endif()
if(CF_FUZZING)
    if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "CF_FUZZING requires Clang libFuzzer")
    endif()
    # Keep coverage-instrumented objects separate from the ordinary test binary.
    add_executable(cf_parser_fuzz tests/host/fuzz_parsers.c ${CF_CORE_SOURCES}
        ${CF_COMPONENT}/third_party/cjson/cf_cJSON.c)
    target_include_directories(cf_parser_fuzz PRIVATE ${CF_COMPONENT}/include
        ${CF_COMPONENT}/src/protocol ${CF_COMPONENT}/third_party/cjson)
    target_compile_definitions(cf_parser_fuzz PRIVATE CJSON_NESTING_LIMIT=16)
    target_link_libraries(cf_parser_fuzz PRIVATE m)
    target_compile_options(cf_parser_fuzz PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
    target_link_options(cf_parser_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
